#+BEGIN_SRC rust-ts
  use memchr::memchr;

  const LINK_TYPES: [&str; 3] = ["https", "http", "file"];

  #[derive(Debug)]
  pub struct Link {
      pub link_type: String,
      pub path: String,
      pub description: Option<String>,
  }

  /// Scannt eine Zeile EINMAL → Org-Links + Plain-Links
  pub fn scan_links(line: &str) -> Vec<Link> {
      let mut result = Vec::new();
      let bytes = line.as_bytes();
      let mut pos = 0usize;

      while pos < bytes.len() {
          // 1) ORG-Link?  [[...][...]]
          if bytes[pos] == b'[' {
              // Check auf "[["
              if pos + 1 < bytes.len() && bytes[pos + 1] == b'[' {
                  if let Some((link, consumed)) = scan_org_at(line, pos) {
                      result.push(link);
                      pos += consumed;
                      continue;
                  }
              }
          }

          // 2) Plain link: type:path
          if bytes[pos].is_ascii_alphabetic() {
              if let Some((link, consumed)) = scan_plain_at(line, pos) {
                  result.push(link);
                  pos += consumed;
                  continue;
              }
          }

          pos += 1;
      }

      result
  }

  //
  // ---------- ORG LINK SCAN -------------
  //
  fn scan_org_at(line: &str, start: usize) -> Option<(Link, usize)> {
      let bytes = line.as_bytes();
      let len = bytes.len();

      // Suche "]["
      let rest = &line[start..];
      let mid_rel = rest.find("][")?;
      let mid = start + mid_rel;

      // Suche "]]"
      let end_rel = rest.find("]]")?;
      let end = start + end_rel;

      // URL und Beschreibung
      let url = &line[start + 2 .. mid];
      let desc = &line[mid + 2 .. end];

      // Typ + Pfad
      let (typ, path) = url.split_once(':').unwrap_or(("unknown", url));

      let link = Link {
          link_type: typ.to_string(),
          path: format!(":{}", path),
          description: Some(desc.to_string()),
      };

      // Zurückgeben + wie viele Zeichen verbraucht wurden
      Some((link, (end - start) + 2))
  }

  //
  // ---------- PLAIN LINK SCAN -------------
  //
  fn scan_plain_at(line: &str, start: usize) -> Option<(Link, usize)> {
      // Beschaffe das restliche Wort
      let rest = &line[start..];

      // Finde Colon
      let cpos = rest.find(':')?;
      let typ = &rest[..cpos];

      if !LINK_TYPES.contains(&typ) {
          return None;
      }

      // Pfad = bis zum nächsten whitespace
      let after = &rest[cpos..];
      let end = after.find(char::is_whitespace).unwrap_or(after.len());

      let path = &after[..end];

      let link = Link {
          link_type: typ.to_string(),
          path: path.to_string(),
          description: None,
      };

      Some((link, cpos + end))
  }

  /// Classify a single Org-mode line according to the rules of
  /// Emacs' `org-element--current-element`.
  ///
  /// This function performs *no parsing* and constructs *no AST nodes*.
  /// Instead, it determines **what kind of Org element *begins* at this line**,
  /// based purely on:
  ///
  ///   1. The current parser mode (`ParserMode`)
  ///      – which encodes structural context such as being inside a list,
  ///        table, drawer, planning block, etc.
  ///
  ///   2. Cheap prefix tests
  ///      – used for headlines, comments, drawer syntax, clock lines,
  ///        planning lines, and other constructs that Org-mode detects
  ///        without regexes.
  ///
  ///   3. The large dispatcher regex `org-element--current-element-re`
  ///      – used by Org-mode to detect LaTeX blocks, drawers, fixed-width
  ///        lines, #+BEGIN_* blocks, CALL lines, keywords, footnotes,
  ///        horizontal rules, and diary sexps.
  ///
  ///   4. Table and list detection
  ///      – applied *after* the big regex, exactly like Org-mode does.
  ///
  ///   5. Default fallback
  ///      – if none of the above match, the line begins a paragraph.
  ///
  /// The returned `OrgLineType` expresses *what parser must handle this line*,
  /// mirroring the control flow of `org-element--current-element`:
  ///
  ///     item        → list item parser
  ///     table-row   → table row parser
  ///     node-property → property parser (not drawers)
  ///     headline    → parse a headline
  ///     section     → the section parser

  /// Validates if a block starting at `start` is well-formed.
  /// Returns Some(end_line) if valid, None if invalid.
  pub fn validate_block(
      lines: &[String],
      start: usize,      // index of the #+BEGIN_ line
      block_name: &str,  // extracted "SRC", "EXAMPLE", …
  ) -> Option<usize> {

      let begin_prefix = format!("#+BEGIN_{}", block_name.to_uppercase());
      let end_prefix   = format!("#+END_{}", block_name.to_uppercase());

      let mut i = start + 1;

      while i < lines.len() {
          let line = lines[i].trim_start();

          // 1. Headline → invalid block
          if line.starts_with('*') {
              return None;
          }

          // 2. END line (case-insensitive in Org)
          if line.to_ascii_uppercase().starts_with(&end_prefix) {
              return Some(i); // valid end line
          }

          i += 1;
      }

      // 3. Reached EOF without END
      None
  }

  if is_begin_block(line) {
      if let Some(end) = validate_block(lines, line_number, block_name) {
          return OrgLineType::ValidBlock { end };
      } else {
          return OrgLineType::Paragraph;
      }
  }

  /// Returns Some(end_line) if drawer is valid, otherwise None.
  pub fn validate_property_drawer(lines: &[String], start: usize) -> Option<usize> {
      let mut i = start + 1;

      while i < lines.len() {
          let raw = lines[i].trim_end();

          let trimmed = raw.trim_start();

          // 1. END?
          if trimmed.eq_ignore_ascii_case(":END:") {
              return Some(i);
          }

          // 2. Every line before END must be a valid property
          // Format: :KEY: VALUE
          if let Some(rest) = trimmed.strip_prefix(':') {
              // must contain another ':' for the key terminator
              if let Some(colon_index) = rest.find(':') {
                  let key = &rest[..colon_index];
                  // key must be alphanumeric, underscore, hyphen (Org-compatible)
                  if key.bytes().all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'-') {
                      // OK property
                  } else {
                      return None; // invalid key
                  }
              } else {
                  return None; // no second colon
              }
          } else {
              return None; // non-property line → invalid drawer
          }

          i += 1;
      }

      // 3. EOF without END → invalid
      None
  }

  if is_properties_begin(line) {
      if let Some(end) = validate_property_drawer(lines, line_number) {
          return OrgLineType::PropertyDrawer { end };
      } else {
          return OrgLineType::Paragraph;
      }
  }

  #[derive(Debug, Clone, Copy, PartialEq, Eq)]
  pub enum ParserMode {
      /// Default state (paragraph or unknown)
      Normal,

      /// Parsing inside an Org list item
      Item,

      /// Inside a table row (table.el or org table)
      TableRow,

      /// Node property line (e.g. `:ID:` before a headline)
      NodeProperty,

      /// Inside a property drawer (`:PROPERTIES:` ... `:END:`)
      PropertyDrawer,

      /// Inside planning block (SCHEDULED, DEADLINE lines after headline)
      Planning,

      /// Section (after a headline)
      Section,

      /// Text before the first headline
      FirstSection,

      /// Top-level comment region
      TopComment,

      /// Inside a comment block / comment section
      CommentBlock,

      /// Inside an inlinetask (rare)
      InlineTask,
  }

  impl Default for ParserMode {
      fn default() -> Self {
          ParserMode::Normal
      }
  }

  /// Whether a mode allows property drawers like Emacs does.
  pub fn property_drawer_allowed(mode: ParserMode) -> bool {
      match mode {
          ParserMode::Planning => false,
          ParserMode::NodeProperty => true,
          ParserMode::PropertyDrawer => true,
          ParserMode::TopComment => true,
          _ => false,
      }
  }
