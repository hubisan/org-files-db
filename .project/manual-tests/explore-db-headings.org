#+BEGIN_SRC sqlite :db ./data/org-files-db.sqlite :colnames yes :column yes :nullvalue 'NULL' :results output
  WITH RECURSIVE tree AS (
    SELECT
      h.id,
      h.parent_id,
      h.file_id,
      h.level,
      h.title,
      printf('%04d', h.id) AS sort_path,
      f.path || ' - 0 ' || h.title AS line
      FROM headings h
           JOIN files f ON f.id = h.file_id
     WHERE h.parent_id IS NULL

     UNION ALL

    SELECT
      h.id,
      h.parent_id,
      h.file_id,
      h.level,
      h.title,
      tree.sort_path || '.' || printf('%04d', h.id) AS sort_path,
      printf('%s- %d %s',
             substr('                                                                                ', 1, h.level * 2),
             h.level,
             h.title
      ) AS line
      FROM headings h
           JOIN tree ON h.parent_id = tree.id
  )
  SELECT line
    FROM tree
   ORDER BY file_id, sort_path;
#+END_SRC
