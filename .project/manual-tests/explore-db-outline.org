** Outline tree

#+BEGIN_SRC sqlite :db ./data/org-files-db.sqlite :colnames yes :column yes :nullvalue 'NULL' :results output
  SELECT
    CASE
    WHEN op.depth = 0 THEN f.path || ' - ' || op.materialized_path || ' ' || h.title
    ELSE printf(
      '%s- %s %s',
      substr('                                                                                ', 1, op.depth * 2),
      op.materialized_path,
      h.title
    )
    END AS line
    FROM outline_path op
         JOIN headings h ON h.id = op.heading_id
         JOIN files f ON f.id = op.file_id
   ORDER BY op.file_id, op.materialized_path;
#+END_SRC
