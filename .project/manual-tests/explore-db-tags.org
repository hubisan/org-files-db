* Tags

#+BEGIN_SRC sqlite :db ./data/org-files-db.sqlite :colnames yes :column yes :nullvalue 'NULL' :results output
    SELECT
      t.heading_id,
      h.level,
      h.title,
      t.tag,
      h.file_id,
      f.path AS file_path
    FROM tags t
    JOIN headings h
      ON h.id = t.heading_id
    JOIN files f
      ON f.id = h.file_id
    ORDER BY
      h.file_id,
      t.heading_id,
      h.level;
#+END_SRC
