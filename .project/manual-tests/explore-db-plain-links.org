* All stored links for plain-links.org

#+BEGIN_SRC sqlite :db ./data/org-files-db.sqlite :colnames yes :column yes :nullvalue 'NULL' :results output
  SELECT l.id,
         h.title AS heading_title,
         l.source_context,
         l.format,
         l.raw,
         l.raw_target,
         l.raw_description,
         l.link_type,
         l.path,
         l.search_option,
         l.path_absolute,
         l.target_file_id,
         l.target_heading_id,
         l.target_custom_id,
         l.target_id
    FROM links l
    JOIN files f ON f.id = l.file_id
    JOIN headings h ON h.id = l.heading_id
   WHERE f.path LIKE '%/plain-links.org'
   ORDER BY l.byte_start;
#+END_SRC
