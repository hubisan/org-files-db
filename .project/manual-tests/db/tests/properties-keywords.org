* Properties

#+BEGIN_SRC sqlite :db ../db/org-files-db.sqlite :colnames yes :column yes :nullvalue 'NULL' :results output
  SELECT
    h.file_id,
    h.id AS heading_id,
    p.key,
    p.value,
    p.source,
    p.append,
    p.line_number
  FROM headings h
  JOIN properties p
    ON p.heading_id = h.id
  ORDER BY
    h.file_id,
    p.line_number,
    p.id;
#+END_SRC

* PROPERTY and CATEGORY keywords

#+BEGIN_SRC sqlite :db ../db/org-files-db.sqlite :colnames yes :column yes :nullvalue 'NULL' :results output
  SELECT
    h.file_id,
    h.id AS heading_id,
    k.keyword,
    k.value,
    k.line_number
  FROM headings h
  JOIN keywords k
    ON k.heading_id = h.id
  ORDER BY
    h.file_id,
    k.line_number,
    k.id;
#+END_SRC
