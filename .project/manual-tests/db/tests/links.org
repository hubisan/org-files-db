Run all in [[./manual-test.org]] to update the database with the files from [[./files/]].

#+BEGIN_SRC sqlite :db ../db/org-files-db.sqlite :colnames yes :column yes :nullvalue 'NULL' :results output
  SELECT l.id,
         h.title AS heading_title,
         l.source_context,
         l.format,
         l.raw,
         l.raw_target,
         l.raw_description,
         l.link_type,
         l.path,
         l.search_option,
         l.path_absolute,
         l.target_file_id,
         l.target_heading_id,
         l.target_custom_id,
         l.target_id
    FROM links l
    JOIN files f ON f.id = l.file_id
    JOIN headings h ON h.id = l.heading_id
   ORDER BY l.id;
#+END_SRC
