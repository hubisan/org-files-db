* All stored links for [[file:files/link-file-resolve/][Link-File-Resolve]] 

#+BEGIN_SRC sqlite :db ./data/org-files-db.sqlite :colnames yes :column yes :nullvalue 'NULL' :results output
  SELECT f.path,
         l.id,
         h.title AS heading_title,
         l.source_context,
         l.format,
         l.raw,
         l.raw_target,
         l.raw_description,
         l.link_type,
         l.path,
         l.search_option,
         l.path_absolute,
         l.target_file_id,
         tf.path AS target_file_path,
         l.target_heading_id,
         l.target_custom_id,
         l.target_id,
         l.resolution_status,
         l.resolution_diagnostic,
         head.parent_id as target_heading_parent_id,
         head.level as target_heading_level
    FROM links l
    JOIN files f ON f.id = l.file_id
    JOIN headings h ON h.id = l.heading_id
    LEFT JOIN headings head ON head.id = l.target_heading_id
    LEFT JOIN files tf ON tf.id = l.target_file_id;
#+END_SRC

  SELECT f.path,
         l.id,
         h.title AS heading_title,
         l.source_context,
         l.format,
         l.raw,
         l.raw_target,
         l.raw_description,
         l.link_type,
         l.path,
         l.search_option,
         l.path_absolute,
         l.target_file_id,
         tf.path AS target_file_path,
         l.target_heading_id,
         l.target_custom_id,
         l.target_id,
         l.resolution_status,
         l.resolution_diagnostic
    FROM links l
    JOIN files f ON f.id = l.file_id
    JOIN headings h ON h.id = l.heading_id
    LEFT JOIN files tf ON tf.id = l.target_file_id
    WHERE l.raw LIKE "%id:%";
