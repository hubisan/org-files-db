  // ------------------------------------------------------------
  // Org Parser Context
  // Tracks parser state: blocks, drawers, headings, and file-level rules.
  // Fully Org-mode compliant.
  // ------------------------------------------------------------

  #[derive(Debug, Clone, Copy, PartialEq, Eq)]
  pub enum BlockType {
      Src,
      Example,
      Export,
      Comment,
  }

  #[derive(Debug)]
  pub struct Context {
      // ------------------------------------------------------------
      // BLOCK STATE
      // ------------------------------------------------------------

      /// Current active block (SRC / EXAMPLE / EXPORT / COMMENT)
      pub block: Option<BlockType>,

      /// True while inside a :PROPERTIES: ... :END: drawer
      pub in_properties_drawer: bool,

      // ------------------------------------------------------------
      // HEADING STATE
      // ------------------------------------------------------------

      /// True after the first heading is seen
      pub has_seen_heading: bool,

      /// True until the first heading appears
      pub before_first_heading: bool,

      // ------------------------------------------------------------
      // PROPERTY DRAWER RULES
      // ------------------------------------------------------------

      /// File-level PROPERTIES drawer is allowed only when:
      /// - before first heading, and
      /// - only comments/empty lines have occurred
      pub file_property_drawer_allowed: bool,

      /// After a heading (and optionally planning lines), exactly one
      /// PROPERTIES drawer is allowed. Disabled once body text appears.
      pub heading_property_drawer_allowed: bool,
  }

  impl Context {
      // ------------------------------------------------------------
      // Constructor
      // ------------------------------------------------------------
      pub fn new() -> Self {
          Context {
              block: None,
              in_properties_drawer: false,

              has_seen_heading: false,
              before_first_heading: true,

              file_property_drawer_allowed: true,
              heading_property_drawer_allowed: false,
          }
      }

      // ------------------------------------------------------------
      // Public: Update state for each line
      // ------------------------------------------------------------
      /// Update the parser state based on the line:
      /// - BEGIN_ blocks
      /// - END_ blocks
      /// - :PROPERTIES:
      /// - :END:
      pub fn update(&mut self, line: &str) {
          let trimmed = line.trim();

          if self.handle_begin_block(trimmed) { return; }
          if self.handle_end_block(trimmed) { return; }

          if self.handle_property_drawer_start(trimmed) { return; }
          if self.handle_property_drawer_end(trimmed) { return; }
      }

      // ------------------------------------------------------------
      // Block START detection
      // ------------------------------------------------------------
      fn handle_begin_block(&mut self, line: &str) -> bool {
          if let Some(rest) = begins_with_ci(line, "#+BEGIN_") {
              let kind = rest.trim().to_ascii_lowercase();

              self.block = match kind.as_str() {
                  "src"     => Some(BlockType::Src),
                  "example" => Some(BlockType::Example),
                  "export"  => Some(BlockType::Export),
                  "comment" => Some(BlockType::Comment),
                  _ => self.block,
              };

              return true;
          }
          false
      }

      // ------------------------------------------------------------
      // Block END detection
      // ------------------------------------------------------------
      fn handle_end_block(&mut self, line: &str) -> bool {
          if let Some(rest) = begins_with_ci(line, "#+END_") {
              let kind = rest.trim().to_ascii_lowercase();

              if let Some(active) = self.block {
                  let matches = match (active, kind.as_str()) {
                      (BlockType::Src,     "src")     => true,
                      (BlockType::Example, "example") => true,
                      (BlockType::Export,  "export")  => true,
                      (BlockType::Comment, "comment") => true,
                      _ => false,
                  };

                  if matches {
                      self.block = None;
                  }
              }
              return true;
          }
          false
      }

      // ------------------------------------------------------------
      // PROPERTIES drawer START
      // ------------------------------------------------------------
      fn handle_property_drawer_start(&mut self, line: &str) -> bool {
          if !line.eq_ignore_ascii_case(":PROPERTIES:") {
              return false;
          }

          // Case 1: file-level drawer
          if self.before_first_heading && self.file_property_drawer_allowed {
              self.in_properties_drawer = true;
              return true;
          }

          // Case 2: heading-level drawer
          if self.heading_property_drawer_allowed {
              self.in_properties_drawer = true;
              return true;
          }

          // Invalid drawer → ignored
          false
      }

      // ------------------------------------------------------------
      // PROPERTIES drawer END
      // ------------------------------------------------------------
      fn handle_property_drawer_end(&mut self, line: &str) -> bool {
          if !line.eq_ignore_ascii_case(":END:") {
              return false;
          }

          if self.in_properties_drawer {
              self.in_properties_drawer = false;
              return true;
          }

          false
      }

      // ------------------------------------------------------------
      // Utility: Are we currently inside a block where parsing is disabled?
      // ------------------------------------------------------------
      pub fn in_ignored_block(&self) -> bool {
          matches!(
              self.block,
              Some(BlockType::Src | BlockType::Example | BlockType::Export | BlockType::Comment)
          )
      }

      // ------------------------------------------------------------
      // Utility: call when a heading is parsed
      // ------------------------------------------------------------
      pub fn note_heading_seen(&mut self) {
          self.has_seen_heading = true;
          self.before_first_heading = false;
          self.heading_property_drawer_allowed = true; // drawer allowed right below heading
      }

      // ------------------------------------------------------------
      // Utility: call when a planning line is parsed
      // ------------------------------------------------------------
      pub fn note_planning_line(&mut self) {
          self.heading_property_drawer_allowed = true;
      }

      // ------------------------------------------------------------
      // Utility: call when normal content appears after heading
      // ------------------------------------------------------------
      pub fn note_body_content(&mut self) {
          // Planning drawer window closes
          self.heading_property_drawer_allowed = false;
      }
  }

  // ------------------------------------------------------------
  // Fast case-insensitive prefix checker
  // ------------------------------------------------------------
  #[inline(always)]
  fn begins_with_ci<'a>(line: &'a str, prefix: &str) -> Option<&'a str> {
      let p = prefix.as_bytes();
      let l = line.as_bytes();

      if l.len() > p.len() && l[..p.len()].eq_ignore_ascii_case(p) {
          Some(&line[p.len()..])
      } else {
          None
      }
  }
